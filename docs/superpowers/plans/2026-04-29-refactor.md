```ts
// hooks/use-scroll-reveal.ts
import { useRef, useEffect, useState } from 'react';

export function useScrollReveal(threshold = 0.1) {
    const ref = useRef<HTMLDivElement>(null);
    const [isVisible, setIsVisible] = useState(false);

    useEffect(() => {
        const observer = new IntersectionObserver(
            ([entry]) => {
                if (entry.isIntersecting) setIsVisible(true);
            },
            { threshold },
        );
        if (ref.current) observer.observe(ref.current);
        return () => observer.disconnect();
    }, [threshold]);

    return { ref, isVisible };
}
```

## Task 6: Canonical CountUp Component

**Files:**
- Create: `components/count-up.tsx`

The existing `CountUp` in `app/page.tsx` uses `setInterval` and scroll reveal.
The existing `CountUp` in `components/about.tsx` uses `requestAnimationFrame` but no scroll reveal.
This canonical version uses `requestAnimationFrame` (smoother) + `useScrollReveal` (only animates when visible).

- [ ] **Step 1: Create the component**

```tsx
// components/count-up.tsx
'use client';

import { useEffect, useState } from 'react';
import { useScrollReveal } from '@/hooks/use-scroll-reveal';

interface CountUpProps {
    end: number;
    suffix?: string;
    duration?: number;
    className?: string;
}

export function CountUp({ end, suffix = '', duration = 2000, className }: CountUpProps) {
    const [count, setCount] = useState(0);
    const { ref, isVisible } = useScrollReveal();

    useEffect(() => {
        if (!isVisible) return;
        let startTime: number | null = null;
        let animationFrame: number;

        const animate = (currentTime: number) => {
            if (!startTime) startTime = currentTime;
            const progress = Math.min((currentTime - startTime) / duration, 1);
            setCount(Math.floor(progress * end));
            if (progress < 1) {
                animationFrame = requestAnimationFrame(animate);
            } else {
                setCount(end);
            }
        };

        animationFrame = requestAnimationFrame(animate);
        return () => cancelAnimationFrame(animationFrame);
    }, [isVisible, end, duration]);

    return (
        <span ref={ref} className={className}>
            {count.toLocaleString()}
            {suffix}
        </span>
    );
}
```
